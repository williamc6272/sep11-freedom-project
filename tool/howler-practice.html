// Create audio var sound = new Howl({ src: ['https://actions.google.com/sounds/v1/alarms/beep_short.ogg'], volume: 0.5, loop: false }); // Play sound function playSound() { sound.play(); } // Pause sound function pauseSound() { sound.pause(); } // Stop sound function stopSound() { sound.stop(); } // Change volume function changeVolume(value) { sound.volume(Number(value)); } // Change playback speed function changeRate(speed) { sound.rate(speed); } // Repeat sound function toggleLoop(enabled) { sound.loop(enabled); } // Fade out over 2 seconds function fadeSound() { sound.fade(sound.volume(), 0, 2000); } playSound();
